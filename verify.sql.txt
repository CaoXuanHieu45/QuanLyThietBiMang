-- 1. Câu lệnh SELECT tìm thiết bị theo ID
SELECT * 
FROM Devices 
WHERE Id = 1;

---- 1. Câu lệnh JOIN giữa bảng Devices và Sites để xem tên thiết bị, IP và tên địa điểm
SELECT 
    d.Name AS TenThiBi, 
    d.IpAddress, 
    s.Name AS TenDiaDiem
FROM Devices d
JOIN Sites s ON d.SiteId = s.Id;

SELECT 
    d.Name AS TenThietBi, 
    d.IpAddress, 
    s.Name AS TenDiaDiem
FROM Devices d
JOIN Sites s ON d.SiteId = s.Id;

---- 2. Xem địa điểm hiện tại của thiết bị
SELECT Id, Name, SiteId 
FROM Devices 
WHERE Id = 22;

---- 3. Xem lịch sử chuyển thiết bị
SELECT * 
FROM DeviceTransfers 
WHERE DeviceId = 22;
